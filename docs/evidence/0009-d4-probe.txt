package ccc

// zz_d4_probe_test.go is the throw-away before/after probe goal 0009's D4 evidence rests on: the
// unflagged conversion path, measured byte-for-byte (over a normalized copy) and report-for-report on
// two trees. It is copied into a throw-away worktree of the pre-0009 revision and run there, then run
// here, and the two outputs are diffed; it is not part of the tree a later revision carries.
//
// Run one tree with:
//
//	CCC_D4_PROBE_OUT=/tmp/ccc-d4-X/probe.txt go test -tags fts5 -count=1 -run TestZZD4Probe \
//	    ./internal/ccc/ -v
//
// It walks every committed fixture document (the file-backed providers' sessions, plus the OpenCode
// document materialized as a store), converts each through every route's writer with fixed options, and
// runs the whole pipeline (Continue, an *installing* run into a scratch home at --verify none, under a
// target binary no host has) per route x document. Nothing it does carries a selection flag: the probe
// measures the *default* path, and the pipeline arm measures the conversion rather than this host's
// installed harnesses (d4ProbeMissingBinary says why).
//
// Every printed line is normalized so two runs of one tree are byte-identical and the two trees' outputs
// are comparable. The generated identities are *canonicalized* rather than replaced by a class
// placeholder: each distinct uuid, Pi entry id and OpenCode id gets a token of its own, the same token
// for every occurrence of the same value, so a folded document still distinguishes one parent from
// another -- a blanket placeholder per class would let a reparented entry fold to the same bytes. The
// pipeline arm measures two artifacts the earlier probe did not: the whole report, folded and digested
// (so every field of it is compared, named by a line or not), and the bytes of the file the installation
// published (so the pipeline is compared on what it actually wrote, not only on the writer's own bytes).
// A target whose artifact is a row inside the harness's own store publishes no file whose bytes are the
// record, and its line says so instead of digesting the store, whose bytes move with the harness's WAL
// handling.

)

// d4ProbeNow is the conversion instant every run is stamped with, so the writers' clocks are a function
// of the document alone.
var d4ProbeNow = time.Date(2026, 7, 1, 12, 0, 0, 0, time.UTC)

// d4ProbeDocument is one committed fixture session as the probe reads it.
type d4ProbeDocument struct {
	name     string
	ref      Ref
	openCode bool
}

// d4ProbeRepoRoot answers the working tree's own root, so paths inside it can be folded away.
func d4ProbeRepoRoot(t *testing.T) string {
	t.Helper()
	root, err := exec.Command("git", "rev-parse", "--show-toplevel").Output()
	if err != nil {
		t.Fatalf("resolving the repository root: %v", err)
	}
	return strings.TrimSpace(string(root))
}

// d4ProbeCodexSessionID reads the session id out of a Codex rollout file name: the trailing UUID after
// the `rollout-<instant>-` prefix.
var d4ProbeCodexSessionID = regexp.MustCompile(
	`([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})\.jsonl$`)

// d4ProbeDocuments answers every committed fixture document, one entry per session, sorted by name.
func d4ProbeDocuments(t *testing.T) []d4ProbeDocument {
	t.Helper()
	root := fixtureRoot(t)
	var docs []d4ProbeDocument
	for _, provider := range []string{"claude", "codex", "pi", "omp"} {
		dir := filepath.Join(root, provider)
		err := filepath.WalkDir(dir, func(path string, entry os.DirEntry, err error) error {
			if err != nil || entry.IsDir() || !strings.HasSuffix(path, ".jsonl") {
				return err
			}
			id := strings.TrimSuffix(entry.Name(), ".jsonl")
			if provider == "codex" {
				match := d4ProbeCodexSessionID.FindStringSubmatch(entry.Name())
				if match == nil {
					t.Errorf("the Codex fixture %s does not end in a session id", path)
					return nil
				}
				id = match[1]
			}
			docs = append(docs, d4ProbeDocument{
				name: provider + ":" + id,
				ref:  Ref{Provider: provider, ID: id},
			})
			return nil
		})
		if err != nil {
			t.Fatalf("walking %s: %v", dir, err)
		}
	}
	docs = append(docs, d4ProbeDocument{
		name:     "opencode:" + openCodeSourceSessionID,
		ref:      openCodeSourceFixtureRef,
		openCode: true,
	})
	sort.Slice(docs, func(i, j int) bool { return docs[i].name < docs[j].name })
	return docs
}

// d4ProbeRead answers one document's read, its configuration and its reader, so the pipeline arm reads
// the same session the writer arm was handed.
func d4ProbeRead(t *testing.T, doc d4ProbeDocument) (Conversation, config.Config, *SourceReader) {
	t.Helper()
	if doc.openCode {
		root := openCodeSourceStore(t, openCodeSourceFixtureDocument(t))
		cfg := openCodeSourceConfig(t, root)
		return readOpenCodeSourceFixture(t, root), cfg, NewSourceReader(cfg)
	}
	cfg := fixtureConfig(t)
	reader := NewSourceReader(cfg)
	conversation, err := reader.Read(context.Background(), doc.ref)
	if err != nil {
		t.Fatalf("reading %s: %v", doc.name, err)
	}
	return conversation, cfg, reader
}

// d4ProbeNormalizer folds everything a conversion mints or discovers out of one measured text, so the
// same document and route print the same line on two runs and two trees.
type d4ProbeNormalizer struct {
	rules []struct {
		re   *regexp.Regexp
		with string
	}
	// idRules name the identities a conversion mints or writes into a document, each with the identity
	// itself in its first group: a uuid (the writers' session ids and the Claude transcript's record
	// uuids), a Pi entry id in a JSON id position, and an OpenCode identity. They are folded by
	// canonicalIDs rather than by a per-class placeholder, so definitions and references agree after
	// folding while two different values cannot collapse into one.
	idRules []*regexp.Regexp
}

func newD4ProbeNormalizer(repoRoot string) *d4ProbeNormalizer {
	normalizer := &d4ProbeNormalizer{}
	add := func(pattern, with string) {
		normalizer.rules = append(normalizer.rules, struct {
			re   *regexp.Regexp
			with string
		}{regexp.MustCompile(pattern), with})
	}
	add(regexp.QuoteMeta(repoRoot), "<repo>")
	add(`/tmp/[^\s"'\\]*`, "<tmp>")
	add(`"file_modified":"[^"]*"`, `"file_modified":"<mtime>"`)
	// Only what a conversion discovers rather than records is folded here: the checkout's own root and
	// scratch paths and the source file's filesystem mtime. Recorded instants are *not* normalized -- the
	// earlier revision replaced every RFC 3339 instant and every numeric `timestamp`/`created`/`updated`
	// with a placeholder, which silently masked a same-length change to a source-recorded time -- so the
	// probe's own control (d4ProbeNormalizerControl) pins both halves: a recorded instant change moves the
	// folded digest, and a minted identity change does not.
	normalizer.idRules = []*regexp.Regexp{
		regexp.MustCompile(`([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})`),
		regexp.MustCompile(`"(?:id|parentId|firstKeptEntryId)":"([0-9a-f]{8})"`),
		regexp.MustCompile(`((?:ses|msg|call|part|prt)_[A-Za-z0-9_]+)`),
	}
	return normalizer
}

// d4ProbeNormalizerControl is the probe's own control for the normalizer, so the normalizer's claims are
// measured in the same run as the lines they fold. Three texts are folded: two that differ only in a
// source-recorded RFC 3339 instant and two that differ only in a source-recorded numeric millisecond
// field must fold to *different* digests (the change has to be visible in the comparison -- the earlier
// revision's blanket instant replacement silently masked it), and two that differ only in a minted uuid
// must fold to the *same* digest (the canonicalization is what makes two runs and two trees comparable
// while a reparented reference still cannot fold to what it folded to before). Each answer is printed,
// so the retained output carries the control beside the lines it vouches for.
func d4ProbeNormalizerControl(t *testing.T, normalizer *d4ProbeNormalizer) {
	t.Helper()
	moved := func(before, after string) bool {
		t.Helper()
		return d4ProbeFoldedFingerprint(normalizer, []byte(before)) !=
			d4ProbeFoldedFingerprint(normalizer, []byte(after))
	}
	instantMoved := moved(`{"at":"2026-05-04T09:15:01.000Z"}`, `{"at":"2026-05-04T09:15:02.000Z"}`)
	millisecondMoved := moved(`{"time":{"created":1750000000000}}`, `{"time":{"created":1750000000001}}`)
	identityMoved := moved(
		`{"id":"0b8f9a12-1111-4444-8888-abcdefabcdef"}`,
		`{"id":"7c2d3e44-2222-4555-9999-fedcbafedcba"}`)
	fmt.Printf("PROBE-CONTROL recorded-instant-change moves-digest=%v\n", instantMoved)
	fmt.Printf("PROBE-CONTROL recorded-millisecond-change moves-digest=%v\n", millisecondMoved)
	fmt.Printf("PROBE-CONTROL minted-identity-change moves-digest=%v\n", identityMoved)
	if !instantMoved {
		t.Error("the normalizer folds two different recorded RFC 3339 instants to the same bytes: a " +
			"source-recorded instant change has to move the digest")
	}
	if !millisecondMoved {
		t.Error("the normalizer folds two different recorded millisecond fields to the same bytes: a " +
			"source-recorded time has to move the digest")
	}
	if identityMoved {
		t.Error("the normalizer does not canonicalize minted identities: two values a conversion minted " +
			"have to fold to the same token, or two runs of one tree are not comparable")
	}
}

// canonicalIDs replaces every identity idRules names with a token of its own: the tokens are numbered by
// first appearance over the text, and every occurrence of the same value gets the same token, so a
// document's own definitions and the references to them still agree after folding while a change that
// moves a reference to another identity -- an entry reparented from one non-null parent to another, a
// parent written where the other tree wrote none -- folds to a different digest, which a per-class
// placeholder could not show. The substitution is positional (the spans the rules matched), so a value
// that happens to occur inside an unrelated longer string is left alone.
func (n *d4ProbeNormalizer) canonicalIDs(text string) string {
	type span struct{ start, end int }
	spans := make([]span, 0, 32)
	values := make([]string, 0, 32)
	for _, rule := range n.idRules {
		for _, at := range rule.FindAllStringSubmatchIndex(text, -1) {
			spans = append(spans, span{start: at[2], end: at[3]})
			values = append(values, text[at[2]:at[3]])
		}
	}
	if len(spans) == 0 {
		return text
	}
	order := make([]int, len(spans))
	for index := range order {
		order[index] = index
	}
	sort.Slice(order, func(i, j int) bool { return spans[order[i]].start < spans[order[j]].start })
	tokens := make(map[string]string, len(spans))
	var builder strings.Builder
	last := 0
	for _, index := range order {
		at, value := spans[index], values[index]
		if at.start < last {
			continue
		}
		token, ok := tokens[value]
		if !ok {
			token = fmt.Sprintf("<id%d>", len(tokens)+1)
			tokens[value] = token
		}
		builder.WriteString(text[last:at.start])
		builder.WriteString(token)
		last = at.end
	}
	builder.WriteString(text[last:])
	return builder.String()
}

// fold folds one measured text: the identities are canonicalized first (their tokens are numbered by
// appearance, so the rules have to see the text whole), then the per-text rules replace what the
// conversion discovered rather than minted.
func (n *d4ProbeNormalizer) fold(text string) string {
	text = n.canonicalIDs(text)
	for _, rule := range n.rules {
		text = rule.re.ReplaceAllString(text, rule.with)
	}
	return text
}

// d4ProbeFingerprint words one measured document: the bytes as written, the folded copy's own length,
// and the sha256 of the folded copy. Both lengths travel because they answer different questions -- the
// written length changes when a document's byte count does, the folded one when the normalized text
// does -- and the digest is the fingerprint a diff compares.
func d4ProbeFingerprint(normalizer *d4ProbeNormalizer, written []byte) string {
	return fmt.Sprintf("written=%d %s", len(written), d4ProbeFoldedFingerprint(normalizer, written))
}

// d4ProbeFoldedFingerprint words the folded copy of a text whose *written* length is not a comparable
// number across two runs: the whole report names the scratch directories one run's own temp paths spell,
// so its JSON length moves with the random spelling of a path although every measured field is the same.
// What is compared is the folded text, whose paths are folded to a fixed token.
func d4ProbeFoldedFingerprint(normalizer *d4ProbeNormalizer, written []byte) string {
	folded := normalizer.fold(string(written))
	digest := sha256.Sum256([]byte(folded))
	return fmt.Sprintf("folded=%d digest=%s", len(folded), hex.EncodeToString(digest[:]))
}

// d4ProbeLine builds one output line: the document and route names are printed as they are (so a diff
// can select one document), and only the measured text is folded.
func d4ProbeLine(normalizer *d4ProbeNormalizer, kind, doc, route, text string) string {
	line := kind + " " + doc + " " + route
	if text != "" {
		line += " " + normalizer.fold(text)
	}
	return line
}

// d4ProbeWriter runs one route's writer over one document and answers its lines: the normalized digest
// of the written bytes, the carried counts, and every non-zero loss row.
func d4ProbeWriter(
	t *testing.T, doc d4ProbeDocument, conversation Conversation, routeName string,
	route targetRoute, normalizer *d4ProbeNormalizer,
) []string {
	t.Helper()
	converted, err := route.Convert(conversation, targetConvertInput{
		CLIVersion: "d4-probe-0.0",
		Provider:   "d4-probe-provider",
		Model:      "d4-probe-model",
		Now:        d4ProbeNow,
	})
	if err != nil {
		return []string{d4ProbeLine(normalizer, "PROBE-WRITE", doc.name, routeName,
			"error="+err.Error())}
	}
	lines := []string{d4ProbeLine(normalizer, "PROBE-WRITE", doc.name, routeName, fmt.Sprintf(
		"%s carried=%+v", d4ProbeFingerprint(normalizer, converted.Data), converted.Carried))}
	for _, loss := range converted.Losses {
		if loss.Count == 0 {
			continue
		}
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-LOSS", doc.name, routeName,
			fmt.Sprintf("%s %s %d", loss.Category, loss.Disposition, loss.Count)))
	}
	return lines
}

// d4ProbePipeline runs the whole conversion pipeline (Continue: read, writer, installation, structural
// read-back, no verification level) over one document and one route, and answers its lines. The run
// *installs* into a scratch home -- a dry run writes no artifact this probe could fingerprint -- and two
// further measurements travel beside the named fields: the whole report, folded and digested, so every
// field of it is compared rather than only the ones a line names, and the bytes of the file the
// installation published, fingerprinted the way the writer arm fingerprints the writer's own bytes. A
// target whose artifact is a row inside the harness's own store publishes no file whose bytes are the
// record (OpenCode: the store moves with the harness's WAL handling), so its line says which artifact it
// is instead of digesting the store.
//
// d4ProbeMissingBinary is the harness name the arm runs with, and it is deliberately one no host has:
// what the arm compares is the conversion and its installation, and a host's installed harness would put
// the host's own answers into the comparison -- measured on one tree, two runs: a Codex app-server
// handshake reported a timeout where its neighbour reported a closed stream, and the report digest moved
// by eleven bytes between the two runs although every printed field agreed. Naming the absent binary
// keeps this arm the conversion's own measurement; the harness's answers about itself (a version, a
// model provider, an importer's refusal) belong to the native checks and to the record's own native
// sections, not to a before/after comparison of the default path.
const d4ProbeMissingBinary = "ccc-no-such-target-harness"

func d4ProbePipeline(
	t *testing.T, doc d4ProbeDocument, cfg config.Config, reader *SourceReader,
	routeName string, normalizer *d4ProbeNormalizer,
) []string {
	t.Helper()
	home := filepath.Join(t.TempDir(), "d4-probe-home")
	result, err := Continue(context.Background(), ContinueOptions{
		Ref:          doc.ref,
		Target:       routeName,
		Config:       cfg,
		Reader:       reader,
		ArchivePath:  cfg.DBPath,
		TargetHome:   home,
		TargetBinary: d4ProbeMissingBinary,
		Verify:       VerifyNone,
		Now:          d4ProbeNow,
		Stderr:       io.Discard,
	})
	lines := []string{d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
		"conversion=%s format=%s read_from=%s dry_run=%v carried=%+v",
		result.Conversion.State, result.Conversion.Format, result.Conversion.ReadFrom,
		result.Conversion.DryRun, result.Conversion.Carried))}
	if err != nil {
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, "error="+err.Error()))
	}
	if result.Conversion.State == "" {
		return lines
	}
	// The whole report: every field of it, whether a line below names the field or not.
	if encoded, encodeErr := json.Marshal(result); encodeErr != nil {
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
			"report_error="+encodeErr.Error()))
	} else {
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
			"report "+d4ProbeFoldedFingerprint(normalizer, encoded)))
	}
	// The artifact the run published.
	switch path := strings.TrimSpace(result.TargetSession.Path); {
	case path == "":
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
			"artifact=none installation=%s", result.Installation.State)))
	case result.Installation.Mechanism == openCodeImportMechanism:
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
			"artifact=store path=%s (the harness's own store holds the record, so its bytes are not "+
				"digested; the states beside this line are the harness's own answer)", path)))
	default:
		if data, readErr := os.ReadFile(path); readErr != nil {
			lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
				"artifact_error="+readErr.Error()))
		} else {
			lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
				"artifact "+d4ProbeFingerprint(normalizer, data)))
		}
	}
	for _, loss := range result.Conversion.Losses {
		if loss.Count == 0 {
			continue
		}
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
			fmt.Sprintf("loss %s %s %d", loss.Category, loss.Disposition, loss.Count)))
	}
	lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
		"installation state=%s mechanism=%s source_check=%s created=%v",
		result.Installation.State, result.Installation.Mechanism,
		result.Installation.SourceCheckState, result.Installation.Created)))
	lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
		"target provider=%s format=%s session=%s cwd=%s binary_found=%v binary_version=%s cli=%s model_provider=%s",
		result.TargetSession.Provider, result.TargetSession.Format, result.TargetSession.SessionID,
		result.TargetSession.Cwd, result.TargetSession.BinaryFound, result.TargetSession.BinaryVersion,
		result.TargetSession.CLIVersion, result.TargetSession.ModelProvider)))
	lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, fmt.Sprintf(
		"verification structure=%s load=%s resume=%s structure_detail=%s",
		result.Verification.Structure.State, result.Verification.Load.State,
		result.Verification.Resume.State, result.Verification.Structure.Detail)))
	if result.Source != nil {
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName,
			fmt.Sprintf("source counts=%+v", result.Source.Counts)))
	}
	for _, warning := range result.Warnings {
		lines = append(lines, d4ProbeLine(normalizer, "PROBE-RUN", doc.name, routeName, "warning "+warning))
	}
	return lines
}

// TestZZD4Probe writes the probe's own lines to CCC_D4_PROBE_OUT; without the variable it skips.
func TestZZD4Probe(t *testing.T) {
	out := os.Getenv("CCC_D4_PROBE_OUT")
	if out == "" {
		t.Skip("CCC_D4_PROBE_OUT is not set: the D4 before/after probe runs only when asked")
	}
	normalizer := newD4ProbeNormalizer(d4ProbeRepoRoot(t))
	d4ProbeNormalizerControl(t, normalizer)
	docs := d4ProbeDocuments(t)
	routes := Targets()
	sort.Strings(routes)

	var lines []string
	for _, doc := range docs {
		conversation, cfg, reader := d4ProbeRead(t, doc)
		for _, routeName := range routes {
			route, err := targetRouteFor(routeName)
			if err != nil {
				t.Fatalf("route %s: %v", routeName, err)
			}
			writerLines := d4ProbeWriter(t, doc, conversation, routeName, route, normalizer)
			runLines := d4ProbePipeline(t, doc, cfg, reader, routeName, normalizer)
			lines = append(lines, writerLines...)
			lines = append(lines, runLines...)
		}
	}
	if err := os.WriteFile(out, []byte(strings.Join(lines, "\n")+"\n"), 0o644); err != nil {
		t.Fatalf("writing %s: %v", out, err)
	}
	t.Logf("wrote %d line(s) to %s", len(lines), out)
}
